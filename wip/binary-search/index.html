Simple binary search implementation

int binarySearch(int[] a, int target) {
    int left = 0;
    int right = a.length - 1;
    while (left <= right) {
        int mid = left + (right - left) / 2;
        if (target < a[mid]) {
            right = mid - 1;
        } else if (target > a[mid]) {
            left = mid + 1;
        } else {
            return mid;
        }
    }
    return -1;
}
